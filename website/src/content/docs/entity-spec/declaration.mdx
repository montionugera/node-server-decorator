# Declaring Entities

Each supported language provides native, idiomatic syntax to declare entities that compile down to the canonical EntitySpec IR.

<Tabs syncKey="lang">
  <TabItem label="TypeScript / Node" icon="seti:typescript">
```typescript
import { Restful, action, id, str, bool, list, obj, entitySpec } from '@montionugera/entity-spec';

class Address {
  @str({ min: 1 }) street!: string;
  @str({ min: 1 }) city!: string;
}

@Restful({ name: 'User', plural: 'users' })
export class User {
  @id() id!: number;
  @str({ format: 'email' }) email!: string;
  @str({ min: 1, max: 100 }) name!: string;
  @bool({ default: true }) active!: boolean;
  @list(str()) tags!: string[];
  @obj(Address) address!: Address;

  @action({ method: 'POST', path: 'reset-password' })
  resetPassword() {}
}

export const userSpec = entitySpec(User);
```

  <TabItem label="Python" icon="seti:python">
```python
from server_decorator_entity import restful, action, Id, str_field, bool_field, list_field, obj_field, entity_spec

class Address:
    street: str = str_field(min_len=1)
    city: str = str_field(min_len=1)

@restful(name="User", plural="users")
class User:
    id: Id[int]
    email: str = str_field(format="email")
    name: str = str_field(min_len=1, max_len=100)
    active: bool = bool_field(default=True)
    tags: list[str] = list_field(str_field())
    address: Address = obj_field(Address)

    @action(method="POST", path="reset-password")
    def reset_password(self):
        pass

user_spec = entity_spec(User)
```

  <TabItem label="Go" icon="seti:go">
```go
package main

import "github.com/montionugera/node-server-decorator/v2/entity"

func NewUserSpec() entity.EntitySpec {
    return entity.Build("User",
        entity.Plural("users"),
        entity.Field("id", entity.TypeInt, entity.Id()),
        entity.Field("email", entity.TypeString, entity.Format("email")),
        entity.Field("name", entity.TypeString, entity.Min(1), entity.Max(100)),
        entity.Field("active", entity.TypeBool, entity.Default(true)),
        entity.Action("reset-password", "POST"),
    )
}
```
