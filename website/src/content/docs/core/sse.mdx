# Real-Time SSE Streaming

`server-decorator` provides a standardized, real-time Server-Sent Events (SSE) streaming transport across TypeScript, Python, and Go. It handles client subscriptions, non-blocking fan-out broadcasting, periodic keep-alive heartbeat pings, and automatic event replay via the HTTP `Last-Event-ID` header.

## Wire Specification

- **MIME Type**: `text/event-stream; charset=utf-8`
- **Response Headers**:
  - `Cache-Control: no-cache, no-transform`
  - `Connection: keep-alive`
  - `X-Accel-Buffering: no` (disables reverse-proxy buffering in Nginx, Envoy, and cloud ingress)
- **Frame Format**:
  ```text
  id: <event-id>
  event: <event-name>
  data: {"symbol":"BTC/USDT","price":65000.5}

  ```
- **Keep-Alive Heartbeat**: Comment line `: ping\n\n` emitted periodically to prevent connection timeouts.

## Usage

  <TabItem label="Go" icon="seti:go">
```go
package main

import (
	"net/http"
	"time"

	"github.com/montionugera/node-server-decorator/v2/go/sse"
)

type MarketTick struct {
	Symbol string  `json:"symbol"`
	Price  float64 `json:"price"`
}

func main() {
	// Broadcaster with 15s keep-alive heartbeat and 500-event ring buffer for reconnect replay
	broadcaster := sse.NewBroadcaster[MarketTick](
		sse.WithHeartbeat(15 * time.Second),
		sse.WithHistoryLimit(500),
	)
	defer broadcaster.Close()

	// Mount SSE endpoint on standard Go 1.22+ ServeMux
	mux := http.NewServeMux()
	mux.Handle("GET /market/ticks/stream", broadcaster)

	// Publish real-time events to all connected clients
	go func() {
		broadcaster.Broadcast(sse.Event[MarketTick]{
			ID:    "tick-1001",
			Event: "tick",
			Data:  MarketTick{Symbol: "BTC/USDT", Price: 65432.10},
		})
	}()

	http.ListenAndServe(":8080", mux)
}
```

```typescript
import http from 'node:http';
import { SSEBroadcaster, type SSEEvent } from '@montionugera/server-decorator';

interface MarketTick {
  symbol: string;
  price: number;
}

const broadcaster = new SSEBroadcaster<MarketTick>({
  heartbeatMs: 15000,
  historyLimit: 500,
});

const server = http.createServer((req, res) => {
  if (req.url?.startsWith('/market/ticks/stream')) {
    broadcaster.handler(req, res);
    return;
  }
  res.statusCode = 404;
  res.end();
});

// Broadcast ticks to all subscribers
broadcaster.broadcast({
  id: 'tick-1001',
  event: 'tick',
  data: { symbol: 'BTC/USDT', price: 65432.10 },
});

server.listen(8080);
```

```python
from server_decorator import AsyncSSEBroadcaster, SSEEvent
from starlette.applications import Starlette
from starlette.responses import StreamingResponse
from starlette.routing import Route

broadcaster = AsyncSSEBroadcaster[dict](
    heartbeat_seconds=15.0,
    history_limit=500,
)

async def stream_endpoint(request):
    last_event_id = request.headers.get("last-event-id")
    return StreamingResponse(
        broadcaster.subscribe(last_event_id=last_event_id),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache, no-transform",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )

app = Starlette(routes=[Route("/market/ticks/stream", stream_endpoint)])

# Broadcast ticks asynchronously
await broadcaster.broadcast(SSEEvent(
    id="tick-1001",
    event="tick",
    data={"symbol": "BTC/USDT", "price": 65432.10},
))
```
