## ตัวอย่างโค้ดหลายภาษา (Polyglot)

เลือกภาษาที่คุณต้องการเพียงครั้งเดียว โค้ดตัวอย่างในทุกหน้าของเอกสารจะเปลี่ยนเป็นภาษานั้นโดยอัตโนมัติ:

<Tabs syncKey="lang">
  <TabItem label="TypeScript / Node" icon="seti:typescript">
```typescript
import { cache, EmitOnSuccess, tracking } from '@montionugera/server-decorator';

export class UserService {
  @cache({ ttl: 60 })
  @tracking()
  async getUser(id: string) {
    return db.users.findById(id);
  }

  @EmitOnSuccess({ topic: 'user.created' })
  async createUser(payload: CreateUserDTO) {
    return db.users.create(payload);
  }
}
```
  </TabItem>
  <TabItem label="Python" icon="seti:python">
```python
from server_decorator import cache, emit_on_success, tracking

class UserService:
    @cache(ttl=60)
    @tracking()
    async def get_user(self, id: str):
        return await db.users.find_by_id(id)

    @emit_on_success(topic="user.created")
    async def create_user(self, payload: CreateUserDTO):
        return await db.users.create(payload)
```
  </TabItem>
  <TabItem label="Go" icon="seti:go">
```go
package main

import "github.com/montionugera/node-server-decorator/v2/entity"

func main() {
    spec := entity.Build("User",
        entity.Field("id", entity.TypeInt, entity.Id()),
        entity.Field("email", entity.TypeString, entity.Format("email")),
    )
}
```
  </TabItem>
  <TabItem label="Nim" icon="seti:nim">
```nim
import server_decorator

restful User:
  id: Id[int]
  email: str {.format: "email".}
  name: str
```
  </TabItem>
</Tabs>
